--------------------------------------------------------------------------------
HYBRID RRF QUERY (parameterized form)
--------------------------------------------------------------------------------
-- $1 = embedding::vector(1024), $2 = plain-text keyword query
WITH vector_ranked AS (
    SELECT survey_id, RANK() OVER (ORDER BY dist) AS rank
    FROM (
        SELECT survey_id, amenity_embedding <=> $1::vector AS dist
        FROM   survey_demo.river_amenity_search_synced
        ORDER BY dist
        LIMIT 40
    ) v
),
keyword_ranked AS (
    SELECT survey_id, RANK() OVER (ORDER BY score) AS rank
    FROM (
        SELECT survey_id,
               amenity_tsv <@> to_bm25query(
                   to_tsvector('english', $2),
                   'survey_demo.river_amenity_bm25_idx'::regclass
               ) AS score
        FROM   survey_demo.river_amenity_bm25
        ORDER BY score
        LIMIT 40
    ) k
)
SELECT s.survey_id,
       s.amenity_sentiment,
       LEFT(s.amenities_comment, 120) AS amenities_comment,
       COALESCE(1.0 / (60 + v.rank), 0) + COALESCE(1.0 / (60 + k.rank), 0) AS rrf_score
FROM   survey_demo.river_amenity_search_synced s
LEFT JOIN vector_ranked  v USING (survey_id)
LEFT JOIN keyword_ranked k USING (survey_id)
WHERE  v.survey_id IS NOT NULL OR k.survey_id IS NOT NULL
ORDER BY rrf_score DESC, s.survey_id
LIMIT 10;
